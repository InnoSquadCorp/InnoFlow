import assert from 'node:assert/strict';
import { readFile, writeFile } from 'node:fs/promises';
const evidence = '/tmp/innoflow-verification-20261018.vkS7JQI8';
const graph = await readFile(`${evidence}/jvm-dependency-graph.log`, 'utf8');
const coordinates = new Set();
for (const line of graph.split('\n')) {
  const match = line.match(/^[ |+\\:-]*([\w.-]+:[\w.-]+):([^\s]+)(?: -> ([^\s]+))?/);
  if (!match || line.includes(' (c)')) continue;
  const resolved = match[3] ?? match[2];
  if (/^[0-9][\w.+-]*$/.test(resolved)) coordinates.add(`${match[1]}:${resolved}`);
  else if (/^[\w.-]+:[\w.-]+:[0-9][\w.+-]*$/.test(resolved)) coordinates.add(resolved);
}
assert.ok(coordinates.size > 10, 'A real resolved graph is required');
const query = coordinate => {
  const [group, artifact, version] = coordinate.split(':');
  return { package: { ecosystem: 'Maven', name: `${group}:${artifact}` }, version };
};
const inputs = [...coordinates].sort();
const queries = [...inputs.map(query), query('log4j:log4j:1.2.17')];
const all = queries.map(() => []);
let pending = queries.map((value, index) => ({ value, index }));
while (pending.length) {
  const response = await fetch('https://api.osv.dev/v1/querybatch', { method: 'POST',
    headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ queries: pending.map(entry => entry.value) }),
    signal: AbortSignal.timeout(30000) });
  assert.equal(response.status, 200); const document = await response.json();
  assert.equal(document.results.length, pending.length);
  const next = [];
  for (const [index, result] of document.results.entries()) {
    assert.equal(result.error, undefined, JSON.stringify(result));
    const entry = pending[index]; all[entry.index].push(...(result.vulns ?? []));
    if (result.next_page_token) next.push({ index: entry.index, value: { ...entry.value, page_token: result.next_page_token } });
  }
  pending = next;
}
const control = all.at(-1); assert.ok(control.length > 0, 'Known-vulnerable external control must be detected');
const matches = inputs.map((coordinate, i) => ({ coordinate, advisories: all[i].map(v => v.id) }));
const report = { checkedAt: new Date().toISOString(), scope: 'all five published Kotlin modules resolved runtimeClasspath only; not build/test/plugin graph or exploitability audit', packages: matches.length,
  affectedPackages: matches.filter(value => value.advisories.length), control: { coordinate: 'log4j:log4j:1.2.17', advisoryCount: control.length }, matches };
await writeFile(`${evidence}/maven-advisories.json`, JSON.stringify(report, null, 2));
console.log(JSON.stringify(report));
